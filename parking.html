    <script>

        const { createApp, ref, computed, onMounted, watch } = Vue;

 

        createApp({

            setup() {

                // 状态数据

                const isTracking = ref(false);

                const startTime = ref('12:00');

                const cycleMinutes = ref(60); // 默认1小时周期

                const bufferMinutes = ref(15); // 默认提前15分钟

                const now = ref(new Date());

                const permissionStatus = ref(Notification.permission);

                

                // 定时器引用

                let timerInterval = null;

 

                // 初始化：从 LocalStorage 读取数据（防刷新丢失）

                onMounted(() => {

                    setNow();

                    const savedState = localStorage.getItem('parkingState');

                    if (savedState) {

                        const parsed = JSON.parse(savedState);

                        isTracking.value = parsed.isTracking;

                        startTime.value = parsed.startTime;

                        cycleMinutes.value = parsed.cycleMinutes;

                        bufferMinutes.value = parsed.bufferMinutes;

                    }

                    

                    // 启动每秒更新

                    timerInterval = setInterval(() => {

                        now.value = new Date();

                        checkAlerts();

                    }, 1000);

                });

 

                // 监听状态变化并保存

                watch([isTracking, startTime, cycleMinutes, bufferMinutes], () => {

                    localStorage.setItem('parkingState', JSON.stringify({

                        isTracking: isTracking.value,

                        startTime: startTime.value,

                        cycleMinutes: cycleMinutes.value,

                        bufferMinutes: bufferMinutes.value

                    }));

                });

 

                // 设置当前时间

                const setNow = () => {

                    const d = new Date();

                    const h = String(d.getHours()).padStart(2, '0');

                    const m = String(d.getMinutes()).padStart(2, '0');

                    startTime.value = `${h}:${m}`;

                };

 

                // 计算逻辑

                const getStartTimeDate = () => {

                    const d = new Date();

                    const [h, m] = startTime.value.split(':');

                    d.setHours(h, m, 0, 0);

                    // 如果设置的时间比现在晚，说明是跨天（虽然停车很少用到，但逻辑要严谨，这里简化处理假设是当天或昨天）

                    if (d > now.value) {

                        d.setDate(d.getDate() - 1);

                    }

                    return d;

                };

 

                const nextBillingDate = computed(() => {

                    const start = getStartTimeDate();

                    const diffMs = now.value - start;

                    const cycleMs = cycleMinutes.value * 60 * 1000;

                    

                    // 已经过了多少个周期

                    const cyclesPassed = Math.floor(diffMs / cycleMs);

                    // 下一个周期的结束时间

                    const nextEnd = new Date(start.getTime() + (cyclesPassed + 1) * cycleMs);

                    return nextEnd;

                });

 

                const nextBillingTime = computed(() => {

                    const d = nextBillingDate.value;

                    return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;

                });

 

                const remainingSeconds = computed(() => {

                    const diff = (nextBillingDate.value - now.value) / 1000;

                    return Math.max(0, Math.floor(diff));

                });

 

                const remainingTimeFormatted = computed(() => {

                    const s = remainingSeconds.value;

                    const m = Math.floor(s / 60);

                    const sec = s % 60;

                    return `${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;

                });

 

                // 是否紧急（进入缓冲期）

                const isUrgent = computed(() => {

                    return remainingSeconds.value < (bufferMinutes.value * 60);

                });

 

                // 操作

                const requestPermission = () => {

                    Notification.requestPermission().then(p => permissionStatus.value = p);

                };

 

                const startTracking = () => {

                    isTracking.value = true;

                    requestPermission();

                };

 

                const stopTracking = () => {

                    isTracking.value = false;

                    localStorage.removeItem('parkingState');

                    setNow(); // 重置为当前时间方便下次

                };

 

                // 报警逻辑

                let hasAlerted = false;

                const checkAlerts = () => {

                    if (!isTracking.value) return;

                    

                    // 如果进入紧急时间，且还没报过警

                    const timeLeft = remainingSeconds.value;

                    const bufferSec = bufferMinutes.value * 60;

 

                    // 这里做一个简单的逻辑：每当进入一个新的周期的最后 buffer 分钟时提醒

                    // 实际上因为 remainingSeconds 是动态计算的，我们只需要判断它是否刚掉入区间

                    

                    // 简单处理：如果剩余时间等于 Buffer 时间（误差2秒内），发送通知

                    if (Math.abs(timeLeft - bufferSec) < 2) {

                        sendNotification();

                    }

                };

 

                const sendNotification = () => {

                    if (Notification.permission === "granted") {

                        new Notification("�� 该挪车了！", {

                            body: `距离下一个计费周期只剩 ${bufferMinutes.value} 分钟，快去省钱！`,

                            icon: "https://cdn-icons-png.flaticon.com/512/2331/2331966.png"

                        });

                        // 播放声音

                        const audio = new Audio('https://assets.mixkit.co/active_storage/sfx/2869/2869-preview.mp3');

                        audio.play().catch(e => console.log('Audio play failed', e));

                    }

                };

 

                return {

                    isTracking, startTime, cycleMinutes, bufferMinutes,

                    setNow, startTracking, stopTracking,

                    remainingTimeFormatted, nextBillingTime, isUrgent,

                    permissionStatus, requestPermission

                };

            }

        }).mount('#app');

    </script>
